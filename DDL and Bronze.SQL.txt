-- 3. فحص تناسق النصوص (Capital / Small) وتنوع قيم عمود room_type في كل المدن
--------------------------------------------------------------------------------
SELECT '3. Room Type Consistency Check' AS Check_Type, 
       LOWER(RTRIM(LTRIM(room_type))) AS Cleaned_Room_Type, 
       COUNT(*) AS Frequency 
FROM (
    SELECT room_type FROM Bronze.amsterdam_weekdays
    UNION ALL SELECT room_type FROM Bronze.amsterdam_weekends
    UNION ALL SELECT room_type FROM Bronze.athens_weekdays
    UNION ALL SELECT room_type FROM Bronze.athens_weekends
    UNION ALL SELECT room_type FROM Bronze.barcelona_weekdays
    UNION ALL SELECT room_type FROM Bronze.barcelona_weekends
    UNION ALL SELECT room_type FROM Bronze.berlin_weekdays
    UNION ALL SELECT room_type FROM Bronze.berlin_weekends
    UNION ALL SELECT room_type FROM Bronze.budapest_weekdays
    UNION ALL SELECT room_type FROM Bronze.budapest_weekends
    UNION ALL SELECT room_type FROM Bronze.lisbon_weekdays
    UNION ALL SELECT room_type FROM Bronze.lisbon_weekends
    UNION ALL SELECT room_type FROM Bronze.london_weekdays
    UNION ALL SELECT room_type FROM Bronze.london_weekends
    UNION ALL SELECT room_type FROM Bronze.paris_weekdays
    UNION ALL SELECT room_type FROM Bronze.paris_weekends
    UNION ALL SELECT room_type FROM Bronze.rome_weekdays
    UNION ALL SELECT room_type FROM Bronze.rome_weekends
    UNION ALL SELECT room_type FROM Bronze.vienna_weekdays
    UNION ALL SELECT room_type FROM Bronze.vienna_weekends
) AS AllCitiesRooms
GROUP BY LOWER(RTRIM(LTRIM(room_type)));
