USE Airbnb_DW;
GO

--------------------------------------------------------------------------------
-- 1. إنشاء جدول الفضة الموحد مع تطبيق معايير التنظيف وتحويل أنواع البيانات
--------------------------------------------------------------------------------
IF OBJECT_ID('Silver.Fact_Airbnb_Cleaned', 'U') IS NOT NULL 
    DROP TABLE Silver.Fact_Airbnb_Cleaned;
GO

CREATE TABLE Silver.Fact_Airbnb_Cleaned (
    listing_sk                  INT IDENTITY(1,1) PRIMARY KEY,
    city_name                   VARCHAR(50) NOT NULL,
    day_type                    VARCHAR(20) NOT NULL,
    listing_id                  INT,
    price_eur                   DECIMAL(10,2),
    room_type                   VARCHAR(50),
    room_shared                 BIT,
    room_private                BIT,
    person_capacity             INT,
    is_superhost                BIT,
    is_multiple_rooms           INT,
    is_business_listing         INT,
    cleanliness_score           DECIMAL(4,2),
    satisfaction_score          DECIMAL(4,2),
    bedrooms_count              INT,
    dist_city_center_km         DECIMAL(6,2),
    dist_metro_km               DECIMAL(6,2),
    attraction_index            DECIMAL(10,2),
    attraction_index_norm       DECIMAL(10,2),
    restaurant_index            DECIMAL(10,2),
    restaurant_index_norm       DECIMAL(10,2),
    longitude                   DECIMAL(10,6),
    latitude                    DECIMAL(10,6)
);
GO
